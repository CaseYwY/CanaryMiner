# Generated from the Gate 1a Makefile test targets (24 CPU targets). CPU only: no GPU, no network, no miner run.
# Every test builds with -Wall -Wformat -Werror=format; the CLI host tests also build with -Werror.
enable_testing()
set(R ${CMAKE_SOURCE_DIR})
if(MSVC)
  # windows.h min/max macros would otherwise break std::max/min in the shared headers (job.h).
  add_compile_definitions(NOMINMAX)
endif()
if(MSVC)
  # MSVC spelling of the GCC flags below (same checks: printf-format mismatches and, for the CLI host tests, warnings are errors).
  set(CANARY_TEST_WARN /W3 /we4473 /we4474 /we4477 /utf-8 /wd4996)
  set(CANARY_OPT3 /O2)
  set(CANARY_PTHREAD)
  set(CANARY_O1 /O1)
  set(CANARY_CXX_X)
  set(CANARY_CLI_FLAGS /O1 /W3 /WX /wd4101 /wd4996 /utf-8 /FI${R}/tests/pearl_cli_stubs/cuda_stub.h)
else()
  set(CANARY_TEST_WARN -Wall -Wformat -Werror=format)
  set(CANARY_OPT3 -O3)
  set(CANARY_PTHREAD -pthread)
  set(CANARY_O1 -O1)
  set(CANARY_CXX_X "-x c++")
  set(CANARY_CLI_FLAGS ${CANARY_O1} -Wall -Wextra -Werror -Wno-unused-variable -include ${R}/tests/pearl_cli_stubs/cuda_stub.h)
endif()
